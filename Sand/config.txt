# --- Rain ---
# A storm begins once the water deficit (peak level minus current level) exceeds start_layers
# full grid layers (1 layer = 128*128 = 16384). It then returns water at drops_per_tick per
# compute dispatch and keeps raining until the deficit is fully repaid -- so however the water
# went missing, and wherever it went, a bigger loss automatically means a longer storm.
#   drops_per_tick = the rain RATE (water voxels returned per dispatch). The one rate knob.
#                    Spread is automatic: this is divided across a live count of rainable
#                    columns, so every column under any cloud gets identical odds.
#   overshoot      = extra rain past the measured deficit, as a fraction. Drops still falling
#                    already count as water, so the level reads restored just before they land
#                    and some then soak into soil rather than pooling -- without headroom every
#                    storm ends a touch early and the level creeps down. 0.2 = 20% extra.
rain.start_layers=2
rain.drops_per_tick=50
rain.overshoot=0.2
rain.darken_delay=6.0

# --- Clouds ---
# A fixed group of `count` clouds drifts along +X at drift_speed and wraps around, each one
# dissolving over edge_fade_dist as it nears a border and re-emerging on the opposite side.
# The whole field shares one "charge" (0..1): it eases toward banked steam before a storm and
# toward the outstanding water deficit during one, driving opacity from min_alpha to max_alpha
# and greying the field as it rises -- so the sky reacts as one mass, not cloud by cloud.
#   charge_saturation = cloud water at which the sky is fully charged.
#   min_alpha         = opacity with no evaporation at all. Set 0 for a totally clear sky.
cloud.count=32
cloud.drift_speed=2.0
cloud.edge_fade_dist=20.0
cloud.charge_saturation=20000.0
cloud.charge_ease_rate=0.02
cloud.min_alpha=0.05
cloud.max_alpha=0.9
cloud.voxel_size=3.0
cloud.edge_threshold_min=0.15
cloud.edge_threshold_max=0.7
cloud.max_steps=32

# --- Black hole ---
# A black hole is a single tracked object (max 8 at once), not a paintable material. Anything
# within `radius` stops running its own physics and follows the hole's orbit field instead.
#   infall       = how much straight-in pull is mixed into an otherwise fully tangential step.
#                  Small values spiral slowly; raise it and matter dives at the centre instead.
#   orbit_speed  = scales a per-tick step chance of speed/sqrt(distance), so inner orbits shear
#                  past outer ones. Above ~3 everything captured moves every tick.
#   horizon      = how far PAST the body's surface matter is destroyed. Measured from the body, not
#                  from the centre, so the region a hole clears always matches the size it looks.
#
# Orbits cover every inclination, not one disk. Each hole owns a fan of `orbit_planes` candidate
# planes and a particle rides whichever most nearly contains it, giving a shell of great circles at
# all angles; roughly half run retrograde.
#   orbit_planes = how many distinct inclinations exist. Low values (3-6) give a few crisp,
#                  obviously separate rings; high values approach a smooth spherical swarm.
#   plane_grip   = how firmly a particle is held in its own plane. 0 smears the orbits into a
#                  diffuse cloud; high values make each ring razor-thin.
#
# The body grows with what it eats: side length 2*level+1, so 1x1x1 -> 3x3x3 -> 5x5x5 -> ...
#   growth_cost  = multiplier on how much it must swallow to reach the next size. At 1.0 a hole
#                  ends up about as big as what it consumed (27 voxels to become 3x3x3). Raise it
#                  to make growth slower and harder to trigger by accident.
#   max_level    = size ceiling. 8 is a 17x17x17 body. Reach grows by one voxel per level.
blackhole.horizon=1
blackhole.radius=28
blackhole.orbit_speed=2.2
blackhole.infall=0.16
blackhole.plane_grip=0.5
blackhole.glow=0.85
blackhole.orbit_planes=16
blackhole.growth_cost=1.0
blackhole.max_level=8
