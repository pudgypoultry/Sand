# --- Rain ---
# A storm begins once the water deficit (peak level minus current level) exceeds start_layers
# full grid layers (1 layer = 128*128 = 16384). It then returns water at drops_per_tick per
# compute dispatch and keeps raining until the deficit is fully repaid -- so however the water
# went missing, and wherever it went, a bigger loss automatically means a longer storm.
#   drops_per_tick = the rain RATE (water voxels returned per dispatch). The one rate knob.
#                    Spread is automatic: this is divided across a live count of rainable
#                    columns, so every column under any cloud gets identical odds.
#   overshoot      = extra rain past the measured deficit, as a fraction. Drops still falling
#                    already count as water, so the level reads restored just before they land
#                    and some then soak into soil rather than pooling -- without headroom every
#                    storm ends a touch early and the level creeps down. 0.2 = 20% extra.
rain.start_layers=2
rain.drops_per_tick=50
rain.overshoot=0.2
rain.darken_delay=6.0

# --- Clouds ---
# A fixed group of `count` clouds drifts along +X at drift_speed and wraps around, each one
# dissolving over edge_fade_dist as it nears a border and re-emerging on the opposite side.
# The whole field shares one "charge" (0..1): it eases toward banked steam before a storm and
# toward the outstanding water deficit during one, driving opacity from min_alpha to max_alpha
# and greying the field as it rises -- so the sky reacts as one mass, not cloud by cloud.
#   charge_saturation = cloud water at which the sky is fully charged.
#   min_alpha         = opacity with no evaporation at all. Set 0 for a totally clear sky.
cloud.count=32
cloud.drift_speed=2.0
cloud.edge_fade_dist=20.0
cloud.charge_saturation=20000.0
cloud.charge_ease_rate=0.02
cloud.min_alpha=0.05
cloud.max_alpha=0.9
cloud.voxel_size=3.0
cloud.edge_threshold_min=0.15
cloud.edge_threshold_max=0.7
cloud.max_steps=32

# --- Black hole ---
# A black hole is a single tracked object (max 8 at once), not a paintable material. Anything
# within `radius` stops running its own physics and follows the hole's orbit field instead.
#   infall       = chance per move that a particle spirals inward instead of continuing around.
#                  Small values spiral slowly; raise it and matter dives at the centre instead.
#                  Below ~0.05 decay gets unreliable and matter can orbit for a very long time.
#   orbit_speed  = scales a per-tick step chance of speed/sqrt(distance), so inner orbits shear
#                  past outer ones. Above ~3 everything captured moves every tick.
#   horizon      = how far PAST the body's surface matter is voided, as a spherical shell. Measured
#                  from the body rather than the centre, so the region a hole clears keeps matching
#                  the size it is drawn at as it grows, and it never sits flush against what it eats.
#
# Orbits cover every inclination, not one disk. Each hole owns a fan of `orbit_planes` candidate
# planes and a particle rides whichever most nearly contains it, giving a shell of great circles at
# all angles; roughly half run retrograde.
#   orbit_planes = how many distinct inclinations exist. Low values (3-6) give a few crisp,
#                  obviously separate rings; high values approach a smooth spherical swarm.
#   plane_grip   = how firmly a particle is held in its own plane. 0 smears the orbits into a
#                  diffuse cloud; high values make each ring razor-thin.
#
# The body is a ball of radius level+0.5, so it grows 1 -> 3 -> 5 -> ... voxels across as it eats.
#   growth_cost  = multiplier on how much it must swallow to reach the next size. KEEP THIS ABOVE
#                  1.0. The void ball at level L holds almost exactly what a cost of 1.0 requires
#                  to reach level L+1 -- so at 1.0 a hole dropped in solid matter pays for its own
#                  next level by expanding and cascades to max_level in a few ticks. At 2.0 the
#                  shell covers half of each level and the rest has to be accreted from orbit, so
#                  growth tracks what actually spirals in. Raise further for slower growth.
#   max_level    = size ceiling. 8 is a body 17 voxels across. Reach grows by one voxel per level.
#
# A hole that captures nothing within its reach starves and evaporates, so one placed in empty space
# does not sit there forever. Capturing anything at all resets the clock -- matter merely orbiting
# counts, not just matter being eaten -- so only a genuinely isolated hole, or one that has already
# cleaned out everything near it, dissipates.
#   starve_grace = dispatches with nothing captured before it starts shedding mass. At sim speed 1
#                  this is roughly frames, so 300 is about five seconds of finding nothing.
#   decay_rate   = mass shed per dispatch once starving. The hole shrinks level by level as this
#                  eats into its total, and is removed once it reaches zero.
blackhole.starve_grace=300
blackhole.decay_rate=1
blackhole.horizon=1
blackhole.radius=28
blackhole.orbit_speed=2.2
blackhole.infall=0.16
blackhole.plane_grip=0.5
blackhole.glow=0.85
blackhole.orbit_planes=16
blackhole.growth_cost=2.0
blackhole.max_level=8

# --- Water shadows ---
# Fraction of light surviving one water voxel on a shadow ray.
#
# Water used to block shadow rays outright like any solid, and that hard binary test -- not the
# surface normal -- was overwhelmingly what made water look jittery. When a surface voxel shuffles
# one cell, a shadow ray from a nearby cell flips blocked/unblocked and that pixel's brightness
# swings by the whole sun term. Measured against a single one-voxel hop, the shading change is 0.253
# at 0.0 (hard block) against 0.012 at 1.0 -- and 0.012 is the floor set by the surface normal alone,
# so at 1.0 the shadow contributes nothing to the jitter at all. That is a factor of 21.
#
# Attenuating per voxel instead of just switching water shadows off keeps it physical: one voxel of
# spray barely dims anything, a deep pool crosses many voxels and still darkens the bed beneath it.
#   0.9 puts the jitter near the floor and still transmits ~0.35 through ten voxels of water.
#   1.0 removes water shadowing entirely -- the absolute floor, if any residual flicker still reads.
#   0.0 restores the old hard-blocking behaviour.
water.shadow_transmit=0.9

# --- Water surface waves ---
# A scrolling wave pattern perturbs the water's shading normal. Cosmetic only -- no geometry moves
# and the voxel silhouette is unchanged. It has no role in hiding jitter any more; shadow_transmit
# above and the wider water normal handle that at source, so these are purely a look.
#
# Deliberately understated. At the defaults the normal tilts a median of 1.8 degrees, 7.4 at the 95th
# percentile, peaking near 12 -- just inside the ~14 degree half-angle of the specular lobe, so even
# a crest only modulates the highlight instead of switching it fully on and off. That ceiling is what
# keeps it reading as calm water. For reference the first version of this ran a median of 5.1 and
# peaked past 31, which is why it looked overdone.
#   wave_strength = how far the normal tilts. 0 gives a flat, glassy surface.
#   wave_scale    = spatial frequency. 0.6 is long broad swell; raise toward 1.0+ for small chop.
#   wave_speed    = how fast crests travel.
# The pattern is domain-warped and passed through a low-frequency envelope, so crests bend rather
# than running in straight parallel lines and some patches stay calmer than others. That envelope
# matters more at low amplitude, not less -- a quiet surface with perfectly uniform ripple reads as
# a texture rather than as water.
water.wave_strength=0.9
water.wave_scale=0.6
water.wave_speed=0.4

# --- Lava ---
# Lava carries one number, "coolness", and its whole life is a function of it. Absorbing water raises
# it, boiling moisture out of soil raises it, and sitting still raises it slowly -- so "cooled because
# it drank" and "cooled because it sat there" are the same quantity instead of two counters that
# could disagree. The visual stage (hottest to coldest) is derived from it, and at 4 x stage_size the
# voxel turns to dark stone.
#
# That one number is also what makes the sharing rules work. Lava touching lava and lava touching
# dark stone are the same one-unit diffusion step, run in whichever direction the gradient points.
# Dark stone holds a HIGH coolness, so contact drains it toward the lava, and it re-melts the moment
# contact drags it back under the solidify threshold.
#   stage_size    = coolness per visual stage. 4x this is where lava becomes dark stone, so raising
#                   it makes lava live longer at every stage.
#   viscosity     = chance per tick that lava attempts to move at all. This is what makes it creep
#                   rather than flow; a one-voxel step cannot carry a speed, so the speed lives in
#                   how often the step is taken. 1.0 makes it move as freely as water.
#   spread_radius = how far it looks sideways for somewhere to fall. Water searches up to 128, which
#                   is what lets water teleport across a basin to level instantly; keeping this small
#                   is most of what makes lava read as thick.
#   water_cool    = coolness gained per water voxel flashed to steam. High, because a bucket of
#                   water should visibly set lava back several stages.
#   moisture_cool = coolness gained per unit of moisture boiled out of sand or dirt.
#   rest_cool_chance = chance per tick to cool while settled. This is the clock that turns a still
#                   lava pool into stone on its own, with no water involved at all.
#   consume_chance   = chance per tick to eat a neighbour it has already dried out completely.
#   ignite_chance    = chance per tick to set adjacent grass alight.
#   dark_stone_dry_chance = chance per tick for dark stone to shed a unit of water. Very low. Drying
#                   never re-melts stone on its own -- it only makes it easier for lava to reclaim,
#                   so a lone slab stays stone forever instead of spontaneously turning back.
lava.stage_size=48
lava.viscosity=0.25
lava.spread_radius=6
lava.water_cool=40
lava.moisture_cool=2
lava.rest_cool_chance=0.01
lava.consume_chance=0.05
lava.ignite_chance=0.2
lava.dark_stone_dry_chance=0.002
